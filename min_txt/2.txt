གྲོས་དང་། ཁྱད་པར་བློ་གྲོས་དང་། འཕེལ་བའི་བློ་གྲོས་དང་། མཐོང་བ་དོན་ཡོད་དང་། ཤིན་ཏུ་ཡང་དག་ཞུགས་དང་། རབ་ཀྱི་རྩལ་གྱིས་རྣམ་པར་གནོན་པ་དང་། དཔེ་མེད་བློ་གྲོས་དང་། ཉི་མའི་སྙིང་པོ་དང་། ས་འཛིན་དག་དང་ཡང་ཐབས་ཅིག་སྟེ། དེ་དག་ལ་སོགས་
པ་བྱང་ཆུབ་སེམས་དཔའ་བརྒྱད་ཁྲི་དག་དང་ཡང་ཐབས་ཅིག་གོ། །ལྷ་རྣམས་ཀྱི་དབང་པོ་བརྒྱ་བྱིན་དང་། འཁོར་ལྷའི་བུ་ཉི་ཁྲི་འདི་ལྟ་སྟེ། ལྷའི་བུ་ཟླ་བ་དང་། ལྷའི་བུ་ཉི་མ་དང་། ཀུན་དུ་དྲི་དང་། རིན་ཆེན་འོད་དང་། ལྷའི་བུ་སྣང་འོད་དང་། དེ་དག་ལ་སོགས་པ་ལྷའི་བུ་ཉི་ཁྲི་དང་ཡང་ཐབས་
ཅིག་གོ། །རྒྱལ་པོ་ཆེན་པོ་བཞི་དང་། འཁོར་ལྷའི་བུ་སུམ་ཁྲི་འདི་ལྟ་སྟེ། རྒྱལ་པོ་ཆེན་པོ་འཕགས་སྐྱེས་པོ་དང་། རྒྱལ་པོ་ཆེན་པོ་མིག་མི་བཟང་དང་། རྒྱལ་པོ་ཆེན་པོ་ཡུལ་འཁོར་སྲུང་དང་། རྒྱལ་པོ་ཆེན་པོ་རྣམ་ཐོས་ཀྱི་བུ་དང་། ལྷའི་བུ་དབང་ཕྱུག་དང་། ལྷའི་བུ་དབང་ཕྱུག་ཆེན་པོ་དང་། 
འཁོར་ལྷའི་བུ་སུམ་ཁྲི་དག་དང་ཡང་ཐབས་ཅིག་གོ། །མི་མཇེད་ཀྱི་བདག་པོ་ཚངས་པ་དང་། འཁོར་ཚངས་རིས་ཀྱི་ལྷའི་བུ་ཁྲི་ཉིས་སྟོང་འདི་ལྟ་སྟེ། ཚངས་པ་གཙུག་ཕུད་ཅན་དང་། ཚངས་པ་སྐར་མའི་འོད་དང་། དེ་དག་ལ་སོགས་པ་ཚངས་རིས་ཀྱི་ལྷའི་བུ་ཁྲི་ཉིས་སྟོང་དང་ཡང་ཐབས་ཅིག་གོ། །ཀླུའི་རྒྱལ་
པོ་བརྒྱད་འདི་ལྟ་སྟེ། ཀླུའི་རྒྱལ་པོ་དགའ་བོ་དང་། ཉེ་དགའ་དང་། རྒྱ་མཚོ་དང་། ནོར་རྒྱས་ཀྱི་བུ་དང་། འཇོག་པོ་དང་། གཟི་ཅན་དང་། མ་དྲོས་པ་དང་། ཀླུའི་རྒྱལ་པོ་ཨུད་པ་ལ་དང་། འཁོར་ཀླུ་བྱེ་བ་བརྒྱ་སྟོང་མང་པོ་དག་དང་ཡང་ཐབས་ཅིག་གོ། །མིའམ་ཅིའི་རྒྱལ་པོ་བཞི་འདི་ལྟ་སྟེ། 
མིའམ་ཅིའི་རྒྱལ་པོ་ལྗོན་པ་དང་། མིའམ་ཅིའི་རྒྱལ་པོ་ཆོས་ཆེན་དང་། མིའམ་ཅིའི་རྒྱལ་པོ་ཆོས་བཟང་དང་། མིའམ་ཅིའི་རྒྱལ་པོ་ཆོས་འཛིན་དང་། འཁོར་མི་འམ་ཅི་བྱེ་བ་བརྒྱ་སྟོང་མང་པོ་དག་དང་ཡང་ཐབས་ཅིག་གོ། །དྲི་ཟའི་རིས་ཀྱི་ལྷའི་བུ་བཞི་འདི་ལྟ་སྟེ། དྲི་ཟ་ཡིད་དུ་འོང་བ་དང་། 
ཡིད་འོང་དབྱངས་དང་། འཇམ་སྙན་དང་། དྲི་ཟ་དབྱངས་སྙན་དང་། འཁོར་དྲི་ཟ་བྱེ་བ་བརྒྱ་སྟོང་མང་པོ་དག་དང་ཡང་ཐབས་ཅིག་གོ། །ལྷ་མ་ཡིན་གྱི་དབང་པོ་བཞི་འདི་ལྟ་སྟེ། ལྷ་མ་ཡིན་གྱི་དབང་པོ་སྟོབས་ཅན་དང་། ཕྲག་རྩུབ་དང་། ཐགས་བཟངས་རིས་དང་། ལྷ་མ་
